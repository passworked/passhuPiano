    for (int a = 0; a < args.Length; a += 2)
    {
        var matches = System.Linq.Enumerable.ToArray(System.Linq.Enumerable.Where(symbols,
            s => System.Text.RegularExpressions.Regex.IsMatch(s, "^__\\d+_" + (string)args[a] + "__param$")));
        if (matches.Length == 0) throw new System.MissingMemberException(method + "." + args[a]);
        foreach (var symbol in matches) b.SetProgramVariable(symbol, args[a + 1]);
    }
    var entries = program.EntryPoints.GetExportedSymbols();
    b.SendCustomEvent(System.Linq.Enumerable.Contains(entries, method) ? method : "__0_" + method);
}

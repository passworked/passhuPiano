// Execute through Unity MCP execute_code, outside Play mode. Creates no project assets.
var checks = new System.Collections.Generic.List<string>();
void Need(bool ok, string label) { if (!ok) throw new System.Exception(label); checks.Add(label); }
MusicMidiPlayer.Editor.Assets.MidiSongAsset Fixture(int[] held, int interval, int velocity)
{
    var s = UnityEngine.ScriptableObject.CreateInstance<MusicMidiPlayer.Editor.Assets.MidiSongAsset>();
    int n = held.Length;
    s.sourceHash = new string('0', 64); s.songId = new string('1',32);
    s.startMs = Enumerable.Range(0,n).Select(i=>i*interval).ToArray();
    s.keyUpMs = Enumerable.Range(0,n).Select(i=>s.startMs[i]+held[i]).ToArray();
    s.releaseMs = (int[])s.keyUpMs.Clone(); s.durationMs = s.releaseMs.Max();
    s.pitch=Enumerable.Repeat(69,n).ToArray(); s.velocity=Enumerable.Repeat(velocity,n).ToArray();
    s.channel=new int[n]; s.trackIndex=new int[n];
    var events=Enumerable.Range(0,n).SelectMany(i=>Enumerable.Range(0,3).Select(k=>new {id=i,kind=k,time=k==0?s.startMs[i]:s.keyUpMs[i]})).OrderBy(e=>e.time).ThenBy(e=>e.id).ThenBy(e=>e.kind).ToArray();
    s.eventMs=events.Select(e=>e.time).ToArray();s.eventKind=events.Select(e=>e.kind).ToArray();s.eventNoteId=events.Select(e=>e.id).ToArray();
    int cp=s.durationMs/1000+1;s.checkpointEventIndex=new int[cp];s.checkpointActiveOffsets=new int[cp+1];
    var ids=new System.Collections.Generic.List<int>();
    for(int c=0;c<cp;c++) { s.checkpointEventIndex[c]=events.Count(e=>e.time<c*1000);ids.AddRange(Enumerable.Range(0,n).Where(i=>s.startMs[i]<c*1000 && s.releaseMs[i]>=c*1000));s.checkpointActiveOffsets[c+1]=ids.Count; }
    s.checkpointActiveNoteIds=ids.ToArray();s.pedalOffsets=new int[4];return s;
}
int[] Styles(MusicMidiPlayer.Editor.Assets.MidiSongAsset s)
{
    byte[] p=MusicMidiPlayer.Editor.Baking.ViolinArticulationBaker.Bake(s);
    return Enumerable.Range(0,s.startMs.Length).Select(n=>((p[n>>2]>>((n&3)*2))&3)-1).ToArray();
}
void Phrase(int count,int held,int interval,int velocity,int expected,string label)
{
    var s=Fixture(Enumerable.Repeat(held,count).ToArray(),interval,velocity);
    try { Need(Styles(s).All(x=>x==expected),label);MusicMidiPlayer.Editor.Baking.MidiSongCodec.ValidateRoundTrip(s); }
    finally { UnityEngine.Object.DestroyImmediate(s); }
}
